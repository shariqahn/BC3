// Sent from En-ROADS to BC3 (once at init time, or when the active language or units is changed).

    "tooltipText": "at this location by ${year}"
}, '*')

// Sent from En-ROADS to BC3 to change map temperature, projection, unit, and/or position.

  tbar: '2.3',
  projection: 'equalEarth',
  lat: 50,

  pitch: 0,

// TOOLTIPS

// 'set-tooltip'

  lon: 75
}, '*')
